enable_language(ASM_MASM)
set(FRAMEWORK "${PROJECT_SOURCE_DIR}")
add_subdirectory("${MT2FW_VENDOR}/minhook" "${CMAKE_BINARY_DIR}/minhook" EXCLUDE_FROM_ALL)
set(IMGUI "${MT2FW_VENDOR}/imgui")
add_library(mt2fw_imgui STATIC "${IMGUI}/imgui.cpp" "${IMGUI}/imgui_draw.cpp"
  "${IMGUI}/imgui_tables.cpp" "${IMGUI}/imgui_widgets.cpp"
  "${IMGUI}/backends/imgui_impl_sdl3.cpp" "${IMGUI}/backends/imgui_impl_opengl3.cpp")
target_include_directories(mt2fw_imgui PUBLIC "${IMGUI}" "${IMGUI}/backends" "${MT2FW_VENDOR}/SDL3/include")
target_link_libraries(mt2fw_imgui PUBLIC opengl32 "${MT2FW_VENDOR}/SDL3/lib/x64/SDL3.lib")
add_library(mt2fw_mt2_native STATIC native_engine.cpp internal/logger.cpp)
target_include_directories(mt2fw_mt2_native PUBLIC internal PRIVATE "${FRAMEWORK}/third_party" "${MT2FW_VENDOR}")
target_compile_options(mt2fw_mt2_native PRIVATE /Gy /utf-8 /MP2)
target_compile_definitions(mt2fw_mt2_native PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
add_library(MT2Framework SHARED runtime.cpp adapter.cpp)
target_include_directories(MT2Framework PRIVATE "${FRAMEWORK}/src/core" "${FRAMEWORK}/src/loader")
target_compile_definitions(MT2Framework PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(MT2Framework PRIVATE /utf-8 /MP2)
target_link_options(MT2Framework PRIVATE /OPT:REF)
target_link_libraries(MT2Framework PRIVATE mt2fw_core mt2fw_mt2_native mt2fw_imgui minhook bcrypt)
add_library(mt2fw_loader SHARED "${FRAMEWORK}/src/loader/loader.cpp"
  "${FRAMEWORK}/src/loader/proxy/wsock32_stubs.asm" "${FRAMEWORK}/src/loader/proxy/wsock32_proxy.def")
target_include_directories(mt2fw_loader PRIVATE "${FRAMEWORK}/include" "${FRAMEWORK}/src/core" "${MT2FW_VENDOR}" "${FRAMEWORK}/src/loader/proxy")
target_compile_definitions(mt2fw_loader PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(mt2fw_loader PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/utf-8>")
target_link_libraries(mt2fw_loader PRIVATE mt2fw_files bcrypt ws2_32)
set_target_properties(mt2fw_loader PROPERTIES OUTPUT_NAME "WSOCK32")
