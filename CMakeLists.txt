cmake_minimum_required(VERSION 3.24)
project(MT2Framework VERSION 0.1.4 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(MT2FW_VENDOR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/vendor" CACHE PATH "Vendored third-party sources")
option(MT2FW_GAME "Build the Windows game adapter and loader" ON)
set(lua_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/lua")
file(GLOB LUA_SOURCES "${lua_SOURCE_DIR}/src/*.c")
list(REMOVE_ITEM LUA_SOURCES "${lua_SOURCE_DIR}/src/lua.c" "${lua_SOURCE_DIR}/src/luac.c")
set(LUA_CPP_SOURCES)
foreach(source IN LISTS LUA_SOURCES)
  get_filename_component(stem "${source}" NAME_WE)
  set(wrapper "${CMAKE_BINARY_DIR}/lua-cpp/${stem}.cpp")
  set(wrapper_prefix "#define LUA_CORE\n#include \"lprefix.h\"\nextern \"C\" {\n#include \"lua.h\"\n#include \"lauxlib.h\"\n#include \"lualib.h\"\n}\n")
  if(stem STREQUAL "ldo")
    file(READ "${source}" lua_do)
    set(budget_anchor "L->nCcalls = oldnCcalls;\n  return lj.status;")
    string(FIND "${lua_do}" "${budget_anchor}" anchor_position)
    if(anchor_position EQUAL -1)
      message(FATAL_ERROR "Pinned Lua budget propagation anchor differs")
    endif()
    string(REPLACE "${budget_anchor}" "L->nCcalls = oldnCcalls;\n  if (lj.status == 7 && lj.previous) luaD_throw(L, 7);\n  return lj.status;" lua_do "${lua_do}")
    file(GENERATE OUTPUT "${wrapper}" CONTENT "${wrapper_prefix}${lua_do}\nextern \"C\" int mt2fw_lua_budget_error(lua_State* L) { lua_pushliteral(L, \"Lua instruction budget exceeded (1000000 per callback)\"); luaD_throw(L, 7); return 0; }\n")
  else()
    file(GENERATE OUTPUT "${wrapper}" CONTENT "${wrapper_prefix}#include \"${source}\"\n")
  endif()
  list(APPEND LUA_CPP_SOURCES "${wrapper}")
endforeach()
# Lua uses C++ exceptions internally so API errors unwind host-owned C++ values.
add_library(mt2fw_lua STATIC ${LUA_CPP_SOURCES})
target_include_directories(mt2fw_lua PUBLIC "${lua_SOURCE_DIR}/src")
add_library(mt2fw_files STATIC src/core/files.cpp)
target_include_directories(mt2fw_files PUBLIC include "${MT2FW_VENDOR}")
add_library(mt2fw_core STATIC src/core/core.cpp src/core/lua_mod.cpp)
target_include_directories(mt2fw_core PUBLIC include "${MT2FW_VENDOR}" PRIVATE src/core)
target_link_libraries(mt2fw_core PUBLIC mt2fw_lua mt2fw_files)
target_compile_definitions(mt2fw_core PRIVATE MT2FW_VERSION="${PROJECT_VERSION}")
add_executable(mt2fw_check src/tools/check.cpp)
target_link_libraries(mt2fw_check PRIVATE mt2fw_core)
set_target_properties(mt2fw_check PROPERTIES OUTPUT_NAME "mt2fw")
if(WIN32)
  target_link_libraries(mt2fw_check PRIVATE bcrypt)
  target_compile_definitions(mt2fw_check PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
endif()
if(MSVC)
  target_compile_options(mt2fw_core PRIVATE /W4 /utf-8 /MP2)
  target_compile_options(mt2fw_check PRIVATE /W4 /utf-8)
endif()
enable_testing()
add_executable(mt2fw_core_tests tests/core_tests.cpp)
target_link_libraries(mt2fw_core_tests PRIVATE mt2fw_core)
add_test(NAME framework_core COMMAND mt2fw_core_tests)
set_tests_properties(framework_core PROPERTIES TIMEOUT 15)
add_executable(mt2fw_docs_probe tests/docs_probe.cpp)
target_link_libraries(mt2fw_docs_probe PRIVATE mt2fw_core)
add_library(mt2fw_example_native SHARED templates/native/plugin.cpp)
target_include_directories(mt2fw_example_native PRIVATE include)
set_target_properties(mt2fw_example_native PROPERTIES OUTPUT_NAME "ExampleNative")
if(MT2FW_GAME)
  if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The MT2 adapter requires Windows x64; use MT2FW_GAME=OFF for core tests")
  endif()
  add_subdirectory(adapters/mt2)
endif()
