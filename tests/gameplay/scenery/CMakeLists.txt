cmake_minimum_required(VERSION 3.24)
project(mt2fw_scenery_checks LANGUAGES CXX)
if(NOT MSVC)
  message(FATAL_ERROR "Scenery native ABI checks require Windows MSVC")
endif()
get_filename_component(FRAMEWORK "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_library(scenery_adapter_compile OBJECT "${FRAMEWORK}/adapters/mt2/gameplay/scenery/scenery.cpp")
target_include_directories(scenery_adapter_compile PRIVATE "${FRAMEWORK}/include" "${FRAMEWORK}/third_party/vendor" "${FRAMEWORK}/adapters/mt2")
target_compile_features(scenery_adapter_compile PRIVATE cxx_std_20)
target_compile_definitions(scenery_adapter_compile PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(scenery_adapter_compile PRIVATE /W4 /utf-8)
