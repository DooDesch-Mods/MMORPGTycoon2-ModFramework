cmake_minimum_required(VERSION 3.24)
project(mt2fw_path_checks LANGUAGES CXX)
if(NOT MSVC)
  message(FATAL_ERROR "Path native ABI checks require Windows MSVC")
endif()
get_filename_component(FRAMEWORK "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_library(path_adapter_compile OBJECT "${FRAMEWORK}/adapters/mt2/gameplay/paths/paths.cpp")
target_include_directories(path_adapter_compile PRIVATE "${FRAMEWORK}/include" "${FRAMEWORK}/third_party/vendor" "${FRAMEWORK}/adapters/mt2")
target_compile_features(path_adapter_compile PRIVATE cxx_std_20)
target_compile_definitions(path_adapter_compile PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(path_adapter_compile PRIVATE /W4 /utf-8)
add_executable(path_history_checks history_tests.cpp)
target_include_directories(path_history_checks PRIVATE "${FRAMEWORK}/adapters/mt2")
target_compile_features(path_history_checks PRIVATE cxx_std_20)
target_compile_options(path_history_checks PRIVATE /W4 /utf-8)
enable_testing()
add_test(NAME path_native_history_transition COMMAND path_history_checks)
