cmake_minimum_required(VERSION 3.24)
project(mt2fw_character_checks LANGUAGES CXX)
get_filename_component(FRAMEWORK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_library(character_adapter_compile OBJECT "${FRAMEWORK_ROOT}/adapters/mt2/gameplay/characters/characters.cpp")
target_compile_features(character_adapter_compile PRIVATE cxx_std_20)
target_include_directories(character_adapter_compile PRIVATE "${FRAMEWORK_ROOT}/include" "${FRAMEWORK_ROOT}/third_party/vendor")
target_compile_definitions(character_adapter_compile PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
add_executable(character_records_tests test_records.cpp)
target_compile_features(character_records_tests PRIVATE cxx_std_20)
target_include_directories(character_records_tests PRIVATE "${FRAMEWORK_ROOT}/third_party/vendor")
enable_testing()
add_test(NAME character_records COMMAND character_records_tests)
