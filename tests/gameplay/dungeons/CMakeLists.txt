cmake_minimum_required(VERSION 3.24)
project(mt2fw_dungeon_checks LANGUAGES CXX)
if(NOT MSVC)
  message(FATAL_ERROR "Dungeon native ABI checks require Windows MSVC")
endif()
get_filename_component(FRAMEWORK "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_executable(dungeon_outcome_checks outcome_tests.cpp)
add_library(dungeon_adapter_compile OBJECT "${FRAMEWORK}/adapters/mt2/gameplay/dungeons/dungeons.cpp")
foreach(target dungeon_outcome_checks dungeon_adapter_compile)
  target_include_directories(${target} PRIVATE "${FRAMEWORK}/include" "${FRAMEWORK}/third_party/vendor" "${FRAMEWORK}/adapters/mt2")
  target_compile_features(${target} PRIVATE cxx_std_20)
  target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(${target} PRIVATE /W4 /utf-8)
endforeach()
enable_testing()
add_test(NAME dungeon_outcomes_and_validation COMMAND dungeon_outcome_checks)
