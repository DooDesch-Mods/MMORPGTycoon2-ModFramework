cmake_minimum_required(VERSION 3.24)
project(mt2fw_actor_checks LANGUAGES CXX)
get_filename_component(FRAMEWORK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_library(actor_adapter_compile OBJECT "${FRAMEWORK_ROOT}/adapters/mt2/gameplay/actors/actors.cpp")
target_compile_features(actor_adapter_compile PRIVATE cxx_std_20)
target_include_directories(actor_adapter_compile PRIVATE "${FRAMEWORK_ROOT}/include" "${FRAMEWORK_ROOT}/third_party/vendor")
target_compile_definitions(actor_adapter_compile PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
add_executable(actor_model_tests test_model.cpp)
target_compile_features(actor_model_tests PRIVATE cxx_std_20)
target_include_directories(actor_model_tests PRIVATE "${FRAMEWORK_ROOT}/third_party/vendor")
enable_testing()
add_test(NAME actor_model COMMAND actor_model_tests)
